# Examples for `help <command>`. A command starts with "name: what it does", each
# example with "- what it does:" and the command on the next line, indented.

ls: list the files in a folder
- Every file, hidden ones too, with sizes and dates:
  ls -la
- The newest first:
  ls -lt
- The biggest first, with readable sizes:
  ls -lSh
- Only the folders:
  ls -d */

cd: go to another folder
- Your user folder:
  cd
- The folder you were in before:
  cd -
- A folder on another drive:
  cd /d/Projects
- The folder you visit most whose name has "app" in it:
  z app

cp: copy files and folders
- A file, to a new name:
  cp notes.txt notes-old.txt
- A folder and everything in it:
  cp -r src backup/src
- Without overwriting what is there:
  cp -n *.png images/

mv: move or rename files
- Rename a file:
  mv draft.md post.md
- Move several files into a folder:
  mv *.log logs/
- Ask before overwriting:
  mv -i report.pdf archive/

rm: delete files and folders
- A file:
  rm old.txt
- A folder and everything in it:
  rm -r build
- Ask before each file:
  rm -i *.tmp

mkdir: make folders
- Several levels at once:
  mkdir -p src/components/ui
- Three folders:
  mkdir docs tests scripts

touch: make an empty file, or update a file's date
- A new file:
  touch .env
- Several at once:
  touch index.html style.css app.js

cat: show or join files
- Show a file:
  cat README.md
- With line numbers:
  cat -n main.rs
- Join two files into a third:
  cat part1.txt part2.txt > all.txt

less: read a long file or output one screen at a time
- A file (q to quit, / to search):
  less server.log
- The output of a command:
  git log | less

head: the first lines of a file
- The first 10 lines:
  head notes.txt
- The first 3:
  head -n 3 data.csv

tail: the last lines of a file
- The last 20 lines:
  tail -n 20 app.log
- Keep showing new lines as they come (Ctrl+C to stop):
  tail -f app.log

grep: find lines that match
- In one file:
  grep error app.log
- In every file of a folder, with line numbers:
  grep -rn TODO src
- Whatever the case:
  grep -i warning build.log
- Lines that don't match:
  grep -v DEBUG app.log
- Only count them:
  grep -c "status=500" access.log

find: look for files
- By name, in this folder and below:
  find . -name "*.rs"
- Folders only:
  find . -type d -name node_modules
- Changed in the last day:
  find . -mtime -1
- Bigger than 100 MB:
  find . -size +100M

sed: edit text as it goes by
- Replace a word in a file, in place:
  sed -i 's/http:/https:/g' config.yml
- Print lines 10 to 20:
  sed -n '10,20p' data.csv
- Delete empty lines:
  sed '/^$/d' notes.txt

awk: work with columns
- The second column of a file:
  awk '{ print $2 }' data.txt
- Columns split by commas:
  awk -F, '{ print $1, $3 }' people.csv
- Add up the third column:
  awk '{ sum += $3 } END { print sum }' sales.txt

sort: sort lines
- Alphabetically:
  sort names.txt
- As numbers, biggest first:
  sort -rn sizes.txt
- By the second column, separated by commas:
  sort -t, -k2 people.csv
- Without duplicates:
  sort -u words.txt

uniq: drop repeated lines (sort first)
- Count how often each line appears:
  sort access.log | uniq -c | sort -rn
- Only the lines that repeat:
  sort emails.txt | uniq -d

wc: count lines, words and bytes
- Lines in a file:
  wc -l app.log
- Lines of code in a project:
  cat src/*.ts | wc -l

cut: keep some columns
- The first column of a CSV:
  cut -d, -f1 people.csv
- Characters 1 to 8 of each line:
  cut -c1-8 hashes.txt

tr: change or delete characters
- To upper case:
  echo hello | tr a-z A-Z
- Remove the Windows line ends:
  tr -d '\r' < windows.txt > unix.txt

tee: show output and save it too
- Keep a copy of a build log:
  cargo build 2>&1 | tee build.log
- Add to the end of a file instead:
  date | tee -a times.txt

xargs: run a command on each line of input
- Delete the files a search finds:
  find . -name "*.tmp" | xargs rm
- Four at a time:
  cat urls.txt | xargs -P 4 -n 1 curl -sO

tree: a folder as a tree
- Two levels deep:
  tree -L 2
- Folders only:
  tree -d
- Without node_modules:
  tree -I node_modules

du: how much space folders take
- Each folder here, readable:
  du -sh *
- The biggest ones first:
  du -s * | sort -rn | head

df: free space on the drives
- Readable sizes:
  df -h

ps: running programs
- Every process:
  ps
- Find one by name:
  ps | grep node

kill: stop a program
- By its process id:
  kill 4242
- A background job of this shell:
  kill %1
- Every process with that name:
  killall node

open: open a file or a link with its app
- This folder in File Explorer:
  open .
- A web page:
  open https://nebula.awizz.space
- A file in the app Windows uses for it:
  open report.pdf

diff: compare files
- Two files, with context:
  diff -u old.txt new.txt
- Side by side:
  diff -y old.txt new.txt

echo: print text
- With a variable:
  echo "Hello, $USERNAME"
- Into a file (> replaces, >> adds):
  echo "PORT=3000" >> .env

printf: print formatted text
- A table row:
  printf "%-10s %5d\n" apples 12
- A number with two decimals:
  printf "%.2f\n" 3.14159

date: the date and time
- Now:
  date
- In ISO format:
  date +%Y-%m-%d
- For a file name:
  date +%Y%m%d-%H%M%S

sleep: wait
- Two seconds:
  sleep 2
- Then run something:
  sleep 60 && echo done

ln: make links
- A symbolic link (needs Developer Mode or an admin shell):
  ln -s ../shared/config.json config.json

which: where a command comes from
- A program on PATH:
  which git
- Every match:
  which -a python

alias: short names for commands
- Make one (put it in ~/.nebularc to keep it):
  alias gs='git status'
- List them:
  alias

history: commands you typed before
- The last 20:
  history | tail -n 20
- Run the last one again:
  !!
- Search them as you type:
  Ctrl+R

jobs: commands running in the background
- Start one, then list them:
  sleep 30 & jobs
- Bring one to the front:
  fg %1
- Wait for all of them:
  wait

export: variables for this shell and the programs it starts
- Set one:
  export NODE_ENV=production
- For one command only:
  NODE_ENV=test npm test

source: run a script in this shell
- Reload your settings:
  source ~/.nebularc
